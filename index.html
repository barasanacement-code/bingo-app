function submitDeposit() { const amountInput = document.getElementById('deposit-amount').value.trim(); const smsInput = document.getElementById('deposit-sms').value.trim(); const parsedAmount = parseFloat(amountInput); if (!amountInput || isNaN(parsedAmount) || parsedAmount <= 0) { alert("እባክዎን ትክክለኛ የብር መጠን ያስገቡ!"); return; } const txRegex = /^[a-zA-Z0-9\s:-]+$/; if (!smsInput || !txRegex.test(smsInput)) { alert("እባክዎን ትክክለኛ የ SMS መልእክት ወይም Transaction ID ብቻ ያስገቡ!"); return; } const now = new Date(); const dateStr = `${now.getMonth()+1}/${now.getDate()}/${now.getFullYear()}`; // Unique Transaction Key መፍጠር const txId = db.ref().child('recharges').push().key; const depositData = { id: txId, userId: userId, userName: userName, method: selectedPaymentMethod, amount: parsedAmount.toFixed(2), sms_text: smsInput, date: dateStr, status: 'PENDING', timestamp: Date.now() }; // 1. ለአድሚኑ Admin Panel ይታይ ዘንድ በ 'recharges' Node ስር መጻፍ db.ref('recharges/' + txId).set(depositData); // 2. ለተጫዋቹ የራሱ የትራንዛክሽን ታሪክ (History) እንዲታየው በ 'transactions/userId' ስር መጻፍ db.ref('transactions/' + userId + '/' + txId).set(depositData); // 3. ወደ Telegram Bot መረጃ መላክ tg.sendData(JSON.stringify({ action: "deposit_request", tx_id: txId, user_id: userId, user_name: userName, method: selectedPaymentMethod, amount: parsedAmount.toFixed(2), sms_text: smsInput })); alert("የገንዘብ ማስገቢያ ጥያቄዎ በትክክል ተልኳል!"); document.getElementById('deposit-amount').value = ''; document.getElementById('deposit-sms').value = ''; }
